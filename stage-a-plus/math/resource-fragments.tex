\subsection*{Assumptions}
\paragraph{asm:continuous} Divisible masses, fixed linear rates and capacities; no fixed start-up costs or stochastic uncertainty.
\subsection*{Equations}
% Linear resource model
\begin{equation}
\max_{\mathbf x\in\mathbb R_{\ge0}^2} z=\mathbf c^{\mathsf T}\mathbf x,\qquad A\mathbf x\le\mathbf a,\quad A=\begin{pmatrix}1&1\\2&1\end{pmatrix},\ \mathbf a=\begin{pmatrix}4\\6\end{pmatrix},\ \mathbf c=\begin{pmatrix}3\\2\end{pmatrix}
\label{eq:lp-primal}
\end{equation}
% Weighted sum of valid resource constraints gives upper bound
\begin{equation}
3x_1+2x_2=(x_1+x_2)+(2x_1+x_2)\le 4+6=10
\label{eq:lp-bound}
\end{equation}
% Primal feasible point meets explicit dual bound
\begin{equation}
\boxed{\mathbf x^*=(2,2)^{\mathsf T},\quad z^*=10},\qquad \boldsymbol\lambda=(1,1)^{\mathsf T}\ge0,\quad A^{\mathsf T}\boldsymbol\lambda=\mathbf c
\label{eq:lp-certificate}
\end{equation}
\paragraph{derive:lp-upper-bound}
Conditions: x>=0; Ax<=a; lambda>=0; A^Tlambda=c.
\begin{enumerate}
\item Multiply resource inequalities by the nonnegative prices1,1 with units currency/h.
\item Sum them; capacities give10currency; coefficient sum equals revenue vector.
\item This direct inequality proof bounds every feasible solution, independent of solver status.
\end{enumerate}
\paragraph{prop:lp-optimum}
Conditions: continuous\_nonnegative; linear\_fixed\_coefficients; capacities4and6.
\begin{enumerate}
\item At(2,2), both capacities are attained and quantities are nonnegative; revenue is10.
\item Every feasible point has revenue<=10 by eq:lp-bound. Thus(2,2)is globally optimal.
\item Any optimal point must make both inequalities tight because both multipliers are strictly positive; the nonsingular2x2 equations then force(2,2), hence uniqueness.
\end{enumerate}
